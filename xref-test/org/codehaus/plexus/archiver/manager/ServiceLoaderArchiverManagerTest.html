122     @Test
123     void reuseArchiver(@TempDir File tempDir) throws Exception {
124         ArchiverManager manager = new ServiceLoaderArchiverManager();
125 
126         Archiver archiver = manager.getArchiver("jar");
127         assertNotNull(archiver);
128 
129         archiver.addDirectory(tempDir);
130 
131         Archiver newArchiver = manager.getArchiver("jar");
132         assertNotNull(newArchiver);
133         assertNotEquals(newArchiver, archiver);
134 
135         assertFalse(newArchiver.getResources().hasNext());
136     }
137 
138     @Test
139     void archiverAddsArchivedFileSetWithoutContainer(@TempDir File tempDir) throws Exception {
140         ArchiverManager manager = new ServiceLoaderArchiverManager();
141         File archiveFile = new File(tempDir, "repacked.zip");
142 
143         Archiver archiver = manager.getArchiver("zip");
144         archiver.setDestFile(archiveFile);
145         archiver.addArchivedFileSet(DefaultArchivedFileSet.archivedFileSet(new File("src/test/resources/test.zip")));
146         archiver.createArchive();
147 
148         try (ZipFile zipFile = new ZipFile(archiveFile)) {
149             assertThat(zipFile.stream().map(ZipEntry::getName)).anyMatch(name -> name.endsWith("ZipUnArchiver.java"));
150         }
151     }
